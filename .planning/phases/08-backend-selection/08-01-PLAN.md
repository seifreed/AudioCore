```python
from typing import Any
from audiocore.backends.registry import BackendRegistry
from audiocore.types import BackendType

class BackendAvailabilityChecker:
    """Check availability of all registered backends.
    
    Provides fast availability checks without making network calls.
    
    Attributes:
        registry: BackendRegistry instance for backend discovery
    
    Example:
        >>> checker = BackendAvailabilityChecker()
        >>> statuses = checker.get_available_backends()
        >>> for status in statuses:
        ...     print(f"{status.backend_type.value}: {status.available}")
        openai: True
        faster_whisper: False (reason: faster-whisper not installed)
    """
    
    def __init__(self, registry: BackendRegistry | None = None):
        self.registry = registry or BackendRegistry()
    
    def get_available_backends(self) -> list[BackendStatus]:
        """Get availability status for all registered backends.
        
        Returns:
            List of BackendStatus for each registered backend.
        
        Note:
            Does not make network calls. Availability check:
            - OpenAI: Checks API key presence in config
            - Faster-whisper: Checks module import + optional model presence
        """
        statuses = []
        for backend_type in self.registry.list_backends():
            status = self.check_backend(backend_type)
            statuses.append(status)
        return statuses
    
    def check_backend(self, backend_type: BackendType) -> BackendStatus:
        """Check availability for a specific backend.
        
        Args:
            backend_type: Backend type to check (not AUTO)
        
        Returns:
            BackendStatus with availability, reason, and suggestion.
        
        Raises:
            ValueError: If backend_type is BackendType.AUTO
        """
        if backend_type == BackendType.AUTO:
            raise ValueError(
                f"Cannot check availability for {backend_type.value}. "
                "AUTO is a selection policy, not a backend type."
            )
        
        # Check if backend is registered
        if backend_type not in self.registry.list_backends():
            return BackendStatus.unavailable_backend(
                backend_type,
                reason="Backend not registered",
                suggestion=f"Register {backend_type.value} backend before use"
            )
        
        # Get backend and check availability
        try:
            backend = self.registry.get_backend(backend_type)
            is_avail = backend.is_available()
            
            if is_avail:
                return BackendStatus.available_backend(backend_type)
            else:
                return BackendStatus.unavailable_backend(
                    backend_type,
                    reason=self._get_unavailability_reason(backend_type),
                    suggestion=self._get_unavailability_suggestion(backend_type)
                )
        except Exception as e:
            # Any exception during availability check means unavailable
            return BackendStatus.unavailable_backend(
                backend_type,
                reason=f"Error checking availability: {str(e)}",
                suggestion=f"Check {backend_type.value} backend configuration"
            )
    
    def is_backend_available(self, backend_type: BackendType) -> bool:
        """Quick availability check without detailed status.
        
        Args:
            backend_type: Backend type to check
        
        Returns:
            True if available, False otherwise
        """
        status = self.check_backend(backend_type)
        return status.available
    
    def _get_unavailability_reason(self, backend_type: BackendType) -> str:
        """Get human-readable reason for unavailability."""
        # Override in subclass or use backend-specific reasons
        reasons = {
            BackendType.OPENAI: "OpenAI API key not configured",
            BackendType.FASTER_WHISPER: "faster-whisper not installed or model not downloaded",
        }
        return reasons.get(backend_type, f"{backend_type.value} backend unavailable")
    
    def _get_unavailability_suggestion(self, backend_type: BackendType) -> str:
        """Get actionable suggestion for making backend available."""
        suggestions = {
            BackendType.OPENAI: "Set OPENAI_API_KEY environment variable or configure in config file",
            BackendType.FASTER_WHISPER: "Install faster-whisper: pip install faster-whisper. Download model: audiocore models download base",
        }
        return suggestions.get(backend_type, f"Configure {backend_type.value} backend")
```

Key design decisions:
- Constructor takes optional registry (for testing with mock registry)
- get_available_backends() iterates all registered backends
- check_backend() returns detailed BackendStatus with reason and suggestion
- is_backend_available() is a convenience method returning just bool
- _get_unavailability_reason/suggestion can be overridden or extended
- No network calls in availability checks (delegates to backend.is_available())
