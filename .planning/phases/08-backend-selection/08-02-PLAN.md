```python
import logging
from typing import Any

from audiocore.backends.availability import BackendAvailabilityChecker, BackendStatus
from audiocore.backends.registry import BackendRegistry
from audiocore.errors import BackendUnavailableError
from audiocore.types import BackendType, SelectionPolicy

logger = logging.getLogger(__name__)


class BackendSelector:
    """Intelligent backend selection with policy support.
    
    Selects the best backend based on availability, policy preferences,
    and explicit user selection.
    
    Attributes:
        checker: BackendAvailabilityChecker for availability status
        registry: BackendRegistry for backend instances
    
    Policy Resolution:
        AUTO: Select fastest available backend
            Priority: CUDA > MPS > CPU > Cloud (for small files)
            Falls back to any available backend
        
        PREFER_LOCAL: Use faster-whisper if available
            Falls back to OpenAI if local unavailable
            Raises error if no backends available
        
        PREFER_CLOUD: Use OpenAI if API key configured
            Falls back to faster-whisper if no API key
            Raises error if no backends available
    
    Explicit Selection:
        If backend=BackendType.OPENAI or FASTER_WHISPER:
        - Bypass policy entirely
        - Use specified backend if available
        - Raise BackendUnavailableError if unavailable
        - backend=BackendType.AUTO invalid for explicit selection
    
    Example:
        >>> selector = BackendSelector()
        >>> backend_type = selector.select(
        ...     backend=BackendType.AUTO,
        ...     preference=SelectionPolicy.PREFER_LOCAL
        ... )
        >>> backend_type
        <BackendType.FASTER_WHISPER: 'faster_whisper'>
    """
    
    def __init__(
        self,
        checker: BackendAvailabilityChecker | None = None,
        registry: BackendRegistry | None = None
    ):
        self.checker = checker or BackendAvailabilityChecker()
        self.registry = registry or BackendRegistry()
    
    def select(
        self,
        backend: BackendType,
        preference: SelectionPolicy = SelectionPolicy.AUTO
    ) -> BackendType:
        """Select appropriate backend based on explicit choice or policy.
        
        Args:
            backend: Backend type preference (OPENAI, FASTER_WHISPER, or AUTO)
            preference: Selection policy for automatic selection
        
        Returns:
            Selected backend type (OPENAI or FASTER_WHISPER, never AUTO)
        
        Raises:
            BackendUnavailableError: If selected or fallback backend unavailable
            ValueError: If backend is AUTO AND both explicit and policy selection fail
        
        Policy Resolution:
            1. If backend is OPENAI or FASTER_WHISPER:
               - Use explicit selection, bypass policy
               - Raise BackendUnavailableError if unavailable
            
            2. If backend is AUTO:
               - Apply policy to available backends
               - AUTO policy: Select fastest available (CUDA > MPS > CPU > Cloud)
               - PREFER_LOCAL: Use faster-whisper if available
               - PREFER_CLOUD: Use OpenAI if API key configured
               - Raise BackendUnavailableError if no backends available
        """
        # Explicit backend selection (bypass policy)
        if backend in (BackendType.OPENAI, BackendType.FASTER_WHISPER):
            return self._select_explicit(backend)
        
        # Automatic selection based on policy
        if backend == BackendType.AUTO:
            return self._select_by_policy(preference)
        
        # This should never happen with valid BackendType enum
        raise ValueError(f"Invalid backend type: {backend}")
    
    def _select_explicit(self, backend: BackendType) -> BackendType:
        """Handle explicit backend selection.
        
        Args:
            backend: Explicitly requested backend type
        
        Returns:
            The same backend type if available
        
        Raises:
            BackendUnavailableError: If backend unavailable
        """
        status = self.checker.check_backend(backend)
        
        if status.available:
            logger.debug(f"Explicit selection: using {backend.value}")
            return backend
        
        # Backend unavailable - raise error with available alternatives
        available = self._get_available_backend_types()
        raise BackendUnavailableError(
            f"Backend {backend.value} is unavailable: {status.reason}",
            context={
                "backend_type": backend.value,
                "reason": status.reason,
                "available_backends": [b.value for b in available],
            },
            suggestions=[
                status.suggestion or f"Configure {backend.value} backend",
                f"Available backends: {', '.join(b.value for b in available)}",
            ],
        )
    
    def _select_by_policy(self, policy: SelectionPolicy) -> BackendType:
        """Select backend based on policy.
        
        Args:
            policy: Selection policy (AUTO, PREFER_LOCAL, PREFER_CLOUD)
        
        Returns:
            Selected backend type based on policy
        
        Raises:
            BackendUnavailableError: If no backends available
        """
        statuses = self.checker.get_available_backends()
        available = [s for s in statuses if s.available]
        unavailable = [s for s in statuses if not s.available]
        
        if not available:
            raise BackendUnavailableError(
                "No transcription backends available",
                context={
                    "unavailable_backends": [
                        {"type": s.backend_type.value, "reason": s.reason}
                        for s in unavailable
                    ],
                },
                suggestions=[
                    f"Configure a backend: {s.backend_type.value} - {s.suggestion}"
                    for s in unavailable
                ],
            )
        
        # Apply policy to select from available backends
        if policy == SelectionPolicy.AUTO:
            return self._apply_auto_policy(available)
        elif policy == SelectionPolicy.PREFER_LOCAL:
            return self._apply_prefer_local_policy(available)
        elif policy == SelectionPolicy.PREFER_CLOUD:
            return self._apply_prefer_cloud_policy(available)
        else:
            raise ValueError(f"Invalid selection policy: {policy}")
    
    def _apply_auto_policy(self, available: list[BackendStatus]) -> BackendType:
        """Select fastest available backend.
        
        Priority for transcription speed:
        1. faster-whisper with CUDA (fastest)
        2. faster-whisper with MPS (Apple Silicon)
        3. OpenAI API (cloud, no GPU needed)
        4. faster-whisper with CPU (slowest)
        
        Note: We don't have GPU info in BackendStatus, so we prioritize:
        - If faster-whisper available: prefer it (assume GPU if available)
        - If only OpenAI available: use it
        - This is a simplification; future enhancement could check GPU
        """
        available_types = {s.backend_type for s in available}
        
        # Prefer faster-whisper if available (assume GPU acceleration)
        if BackendType.FASTER_WHISPER in available_types:
            logger.debug("AUTO policy: selecting faster_whisper (prefer local)")
            return BackendType.FASTER_WHISPER
        
        # Use OpenAI if available
        if BackendType.OPENAI in available_types:
            logger.debug("AUTO policy: selecting openai (cloud fallback)")
            return BackendType.OPENAI
        
        # Should not reach here if available is non-empty
        raise BackendUnavailableError(
            "No suitable backend found for AUTO policy",
            context={"available_backends": [s.backend_type.value for s in available]},
            suggestions=["Configure at least one backend"],
        )
    
    def _apply_prefer_local_policy(self, available: list[BackendStatus]) -> BackendType:
        """Select local backend if available, else cloud.
        
        Args:
            available: List of available backend statuses
        
        Returns:
            faster_whisper if available, else openai
        """
        available_types = {s.backend_type for s in available}
        
        if BackendType.FASTER_WHISPER in available_types:
            logger.debug("PREFER_LOCAL policy: selecting faster_whisper")
            return BackendType.FASTER_WHISPER
        
        if BackendType.OPENAI in available_types:
            logger.debug("PREFER_LOCAL policy: falling back to openai")
            return BackendType.OPENAI
        
        raise BackendUnavailableError(
            "No backends available for PREFER_LOCAL policy",
            context={"available_backends": [s.backend_type.value for s in available]},
            suggestions=["Configure faster_whisper or openai backend"],
        )
    
    def _apply_prefer_cloud_policy(self, available: list[BackendStatus]) -> BackendType:
        """Select cloud backend if available, else local.
        
        Args:
            available: List of available backend statuses
        
        Returns:
            openai if available, else faster_whisper
        """
        available_types = {s.backend_type for s in available}
        
        if BackendType.OPENAI in available_types:
            logger.debug("PREFER_CLOUD policy: selecting openai")
            return BackendType.OPENAI
        
        if BackendType.FASTER_WHISPER in available_types:
            logger.debug("PREFER_CLOUD policy: falling back to faster_whisper")
            return BackendType.FASTER_WHISPER
        
        raise BackendUnavailableError(
            "No backends available for PREFER_CLOUD policy",
            context={"available_backends": [s.backend_type.value for s in available]},
            suggestions=["Configure openai or faster_whisper backend"],
        )
    
    def _get_available_backend_types(self) -> list[BackendType]:
        """Get list of currently available backend types.
        
        Returns:
            List of available BackendType enum values
        """
        statuses = self.checker.get_available_backends()
        return [s.backend_type for s in statuses if s.available]
    
    def get_backend_instance(
        self,
        backend: BackendType,
        preference: SelectionPolicy = SelectionPolicy.AUTO
    ) -> "TranscriptionBackend":
        """Convenience method to select backend and get instance.
        
        Args:
            backend: Backend type (OPENAI, FASTER_WHISPER, or AUTO)
            preference: Selection policy for automatic selection
        
        Returns:
            Selected backend instance
        
        Raises:
            BackendUnavailableError: If no backend available
        """
        selected_type = self.select(backend, preference)
        return self.registry.get_backend(selected_type)
```

```python
def select_backend(
    backend: BackendType = BackendType.AUTO,
    preference: SelectionPolicy = SelectionPolicy.AUTO,
    selector: BackendSelector | None = None
) -> BackendType:
    """Convenience function for backend selection.
    
    Creates a BackendSelector instance (or uses provided one) and
    selects appropriate backend based on explicit choice or policy.
    
    Args:
        backend: Backend type preference (OPENAI, FASTER_WHISPER, or AUTO)
        preference: Selection policy for automatic selection
        selector: Optional BackendSelector instance (for testing)
    
    Returns:
        Selected backend type (OPENAI or FASTER_WHISPER)
    
    Raises:
        BackendUnavailableError: If selected or fallback backend unavailable
    
    Example:
        >>> from audiocore.backends import select_backend
        >>> from audiocore.types import BackendType, SelectionPolicy
        >>> backend_type = select_backend(
        ...     backend=BackendType.AUTO,
        ...     preference=SelectionPolicy.PREFER_LOCAL
        ... )
        >>> backend_type
        <BackendType.FASTER_WHISPER: 'faster_whisper'>
    """
    selector_instance = selector or BackendSelector()
    return selector_instance.select(backend, preference)


def get_backend(
    backend: BackendType = BackendType.AUTO,
    preference: SelectionPolicy = SelectionPolicy.AUTO,
    selector: BackendSelector | None = None
) -> "TranscriptionBackend":
    """Convenience function to select and get backend instance.
    
    Creates a BackendSelector instance (or uses provided one) and
    returns the backend instance ready for transcription.
    
    Args:
        backend: Backend type preference (OPENAI, FASTER_WHISPER, or AUTO)
        preference: Selection policy for automatic selection
        selector: Optional BackendSelector instance (for testing)
    
    Returns:
        Selected backend instance
    
    Raises:
        BackendUnavailableError: If selected or fallback backend unavailable
    
    Example:
        >>> from audiocore.backends import get_backend
        >>> from audiocore.types import BackendType
        >>> backend = get_backend(backend=BackendType.OPENAI)
        >>> backend.is_available()
        True
    """
    selector_instance = selector or BackendSelector()
    return selector_instance.get_backend_instance(backend, preference)
```
