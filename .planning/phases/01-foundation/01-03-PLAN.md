## Verification Commands

```bash
# Import all models
python -c "from audiocore.models import Segment, MediaInfo, TranscriptionOptions, TranscriptionResult"

# Verify strict validation
python -c "
from pydantic import ValidationError
from audiocore.models import Segment
try:
    Segment(start_time=10.0, end_time=5.0, text='test')
    print('FAIL: Should have raised ValidationError')
except ValidationError as e:
    print('OK: Cross-field validation works')
"

# Verify defaults
python -c "
from audiocore.models import TranscriptionOptions
from audiocore.types import BackendType, ModelSize, OutputFormat
opts = TranscriptionOptions()
assert opts.backend == BackendType.AUTO
assert opts.model_size == ModelSize.BASE
assert opts.output_format == OutputFormat.TEXT
print('Defaults verified')
"

# Verify serialization
python -c "
from audiocore.models import Segment
s = Segment(start_time=0.0, end_time=5.0, text='test')
d = s.model_dump()
s2 = Segment.model_validate(d)
assert s.text == s2.text
print('Serialization verified')
"

# Run tests
pytest tests/unit/models/ -v
```
