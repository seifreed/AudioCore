<action>
Create src/audiocore/config/merger.py with:
- merge_configs(defaults, toml, env, cli) -> dict[str, Any]:

    f.flush()
    
    # Set env var to override
    os.environ['AUDIOCORE_BACKEND'] = 'openai'

    # env should override toml

    del os.environ['AUDIOCORE_BACKEND']
    os.unlink(f.name)

print('Full priority chain works')
"

# Test with CLI override
python -c "
from audiocore.config import load_config
from audiocore.types import ModelSize
import tempfile

with tempfile.NamedTemporaryFile(mode='w', suffix='.toml', delete=False) as f:
    f.write('[backend]\\nmodel_size = \"small\"\\n')
    f.flush()
    
    # CLI should override toml
    config = load_config(
        config_path=f.name,
        cli_overrides={'model_size': 'large'}
    )
    
    assert config.model_size == ModelSize.LARGE
    print(f'CLI override works: model_size = {config.model_size}')
"

from pydantic import SecretStr

assert 'sk-super-secret-key' not in str(config)
