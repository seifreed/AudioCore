```python
def _get_model(self, options: TranscriptionOptions) -> "WhisperModel":
    """Get or create WhisperModel (lazy loading).
    
    Downloads model if not cached. Reuses cached model across calls.
    
    Args:
        options: TranscriptionOptions with model_size, device, compute_type.
    
    Returns:
        WhisperModel instance ready for transcription.
    
    Raises:
        BackendUnavailableError: If model download fails.
        TranscriptionError: If model loading fails.
    """
    # Get model name from options
    model_name = options.model_size.value
    
    # Determine device (from config, options, or auto-detect)
    device = self._config.device if self._config and self._config.device else None
    if device is None:
        device = get_best_device()
    
    # Determine compute type (from config)
    compute_type = self._config.compute_type if self._config else "default"
    
    # Check if model needs to be reloaded (different size/device/compute_type)
    # If already loaded and same settings, return cached model
    # Otherwise, download/load new model
    
    try:
        # Download model if not cached
        model_manager = self._get_model_manager()
        if not model_manager.is_model_downloaded(model_name):
            model_manager.download_model(model_name)
        
        model_path = model_manager.get_model_path(model_name)
        
        # Import here to avoid import errors in is_available()
        from faster_whisper import WhisperModel
        
        # Create or reuse model
        if self._model is None or self._current_model != model_name:
            self._model = WhisperModel(
                str(model_path),
                device=device,
                compute_type=compute_type,
            )
            self._current_model = model_name
        
        return self._model
        
    except ImportError as e:
        raise BackendUnavailableError(
            "faster-whisper not installed",
            context={"backend": "faster_whisper", "error": str(e)},
            suggestions=[
                "Install faster-whisper: pip install faster-whisper",
                "Verify Python environment has faster-whisper package",
            ],
        ) from e
    except Exception as e:
        # Map to TranscriptionError with context
        raise TranscriptionError(
            f"Failed to load model {model_name}: {e}",
            context={"model": model_name, "device": device},
            suggestions=[
                "Check model is downloaded correctly",
                "Try smaller model size (tiny, base)",
                "Check available disk space",
            ],
        ) from e
```

```python
def transcribe(
    self, audio_path: Path | str, options: TranscriptionOptions
) -> TranscriptionResult:
    """Transcribe audio using faster-whisper.
    
    Args:
        audio_path: Path to audio file.
        options: Transcription configuration.
    
    Returns:
        TranscriptionResult with segments and metadata.
    
    Raises:
        InvalidInputError: If audio file not found.
        BackendUnavailableError: If model not available.
        TranscriptionError: If transcription fails.
    """
    import time
    from pathlib import Path
    
    # Validate file exists
    audio_path = Path(audio_path)
    if not audio_path.exists():
        raise InvalidInputError(
            f"Audio file not found: {audio_path}",
            context={"file_path": str(audio_path), "backend": "faster_whisper"},
            suggestions=[
                "Verify the file path is correct",
                "Check the file exists",
            ],
        )
    
    # Get model (lazy load)
    model = self._get_model(options)
    
    # Build transcription parameters
    transcribe_params: dict[str, Any] = {}
    if self._config:
        # Language
        if self._config.language:
            transcribe_params["language"] = self._config.language
        elif options.language:
            transcribe_params["language"] = options.language
        
        # Decoding parameters
        transcribe_params["beam_size"] = self._config.beam_size
        transcribe_params["best_of"] = self._config.best_of
        transcribe_params["patience"] = self._config.patience
        transcribe_params["temperature"] = self._config.temperature
        transcribe_params["compression_ratio_threshold"] = self._config.compression_ratio_threshold
        transcribe_params["log_prob_threshold"] = self._config.log_prob_threshold
        transcribe_params["no_speech_threshold"] = self._config.no_speech_threshold
        transcribe_params["condition_on_previous_text"] = self._config.condition_on_previous_text
        transcribe_params["word_timestamps"] = self._config.word_timestamps
        transcribe_params["vad_filter"] = self._config.vad_filter
        
        if self._config.initial_prompt:
            transcribe_params["initial_prompt"] = self._config.initial_prompt
    
    start_time = time.time()
    
    try:
        # Transcribe
        segments_iter, info = model.transcribe(str(audio_path), **transcribe_params)
        
        # Convert segments iterator to list
        segments_list = []
        for segment in segments_iter:
            segments_list.append(
                Segment(
                    start_time=segment.start,
                    end_time=segment.end,
                    text=segment.text.strip(),
                )
            )
        
        # Get duration from info
        duration = info.duration if hasattr(info, "duration") else 0.01
        if duration == 0 and segments_list:
            duration = segments_list[-1].end_time
        
        # Build result
        media_info = MediaInfo(
            duration=duration,
            format=audio_path.suffix.lstrip("."),
        )
        
        result = TranscriptionResult(
            segments=segments_list,
            media_info=media_info,
            config_used=options,
            duration_seconds=time.time() - start_time,
            backend_used=BackendType.FASTER_WHISPER,
        )
        
        return result
        
    except Exception as e:
        # Map all exceptions to TranscriptionError
        raise TranscriptionError(
            f"Faster-whisper transcription failed: {e}",
            context={
                "backend": "faster_whisper",
                "file_path": str(audio_path),
                "model": options.model_size.value,
            },
            suggestions=[
                "Check audio file format",
                "Try with different model size",
                "Check available memory",
                "Verify audio is valid",
            ],
        ) from e
```

Key implementation details:
1. Validate audio file exists before transcription (InvalidInputError)
2. Build transcribe_params from FasterWhisperConfig
3. Convert faster-whisper segments iterator to our Segment list
4. Extract duration from info object (fallback to last segment end_time)
5. Create MediaInfo from file extension and duration
6. Map ALL exceptions to TranscriptionError with context
7. Use config for parameters, options as fallback for language
